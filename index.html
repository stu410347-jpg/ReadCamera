async function onScanSuccess(decodedText, decodedResult) { if (isProcessing) return; isProcessing = true; const statusEl = document.getElementById('status'); const resultEl = document.getElementById('scanned-result'); resultEl.textContent = decodedText; statusEl.textContent = '掃描成功，正在寫入試算表...'; statusEl.style.color = '#d97706'; try { // 使用 URLSearchParams 以 Form 格式發送，避開 Google CORS 限制 const formData = new URLSearchParams(); formData.append('qrContent', decodedText); await fetch(GOOGLE_SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: formData.toString() }); statusEl.textContent = '✅ 已成功寫入 Google 試算表！'; statusEl.style.color = 'green'; } catch (error) { console.error('寫入失敗：', error); statusEl.textContent = '❌ 寫入試算表失敗，請重試。'; statusEl.style.color = 'red'; } setTimeout(() => { isProcessing = false; statusEl.textContent = '請對準下一個 QR Code...'; statusEl.style.color = '#333'; }, 3000); }
